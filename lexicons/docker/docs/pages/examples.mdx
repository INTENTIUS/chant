## Web app with database

API service + PostgreSQL + named volume.

```typescript
import { Service, Volume, env } from "@intentius/chant-lexicon-docker";

export const db = new Service({
  image: "postgres:16-alpine",
  environment: {
    POSTGRES_DB: "myapp",
    POSTGRES_USER: "myapp",
    POSTGRES_PASSWORD: env("DB_PASSWORD", { required: true }),
  },
  volumes: ["pgdata:/var/lib/postgresql/data"],
  restart: "unless-stopped",
  healthcheck: {
    test: ["CMD-SHELL", "pg_isready -U myapp"],
    interval: "10s",
    timeout: "5s",
    retries: 5,
  },
});

export const pgdata = new Volume({});

export const api = new Service({
  image: env("API_IMAGE", { default: "myapp:latest" }),
  ports: ["8080:8080"],
  environment: {
    DATABASE_URL: "postgresql://myapp:${DB_PASSWORD}@db:5432/myapp",
  },
  depends_on: ["db"],
  restart: "unless-stopped",
});
```

## Nginx reverse proxy

```typescript
import { Service, DockerConfig } from "@intentius/chant-lexicon-docker";

export const nginxConf = new DockerConfig({ file: "./nginx.conf" });

export const proxy = new Service({
  image: "nginx:1.25-alpine",
  ports: ["80:80", "443:443"],
  configs: [{ source: "nginxConf", target: "/etc/nginx/nginx.conf" }],
  depends_on: ["api"],
  restart: "unless-stopped",
});
```

## Multi-stage build

The build stage copies the manifests and installs before copying the source, so the install layer is reused until the manifests change. The runtime stage copies only the built app and its production dependencies, and runs as `node`.

```typescript
import { Dockerfile, Service } from "@intentius/chant-lexicon-docker";

export const app = new Dockerfile({
  stages: [
    {
      from: "node:20-alpine",
      as: "build",
      steps: [
        { workdir: "/app" },
        { copy: "package.json package-lock.json ./" },
        { run: "npm ci" },
        { copy: "src/ ./src/" },
        { run: "npm run build" },
        { run: "npm prune --omit=dev" },
      ],
    },
    {
      from: "node:20-alpine",
      steps: [
        { workdir: "/app" },
        { copy: "--from=build /app/node_modules ./node_modules" },
        { copy: "--from=build /app/dist ./dist" },
        { user: "node" },
        { expose: 3000 },
        { cmd: ["node", "dist/index.js"] },
      ],
    },
  ],
});

export const api = new Service({
  build: { context: ".", dockerfile: "Dockerfile.app" },
  ports: ["3000:3000"],
  restart: "unless-stopped",
});
```

`Dockerfile.app`:

```dockerfile
FROM node:20-alpine AS build
WORKDIR /app
COPY package.json package-lock.json ./
RUN npm ci
COPY src/ ./src/
RUN npm run build
RUN npm prune --omit=dev
FROM node:20-alpine
WORKDIR /app
COPY --from=build /app/node_modules ./node_modules
COPY --from=build /app/dist ./dist
USER node
EXPOSE 3000
CMD ["node","dist/index.js"]
```

To share stages or steps between Dockerfiles, see [Sharing stages and steps between Dockerfiles](/chant/lexicons/docker/dockerfiles/#sharing-stages-and-steps-between-dockerfiles).

## Team-wide labels

```typescript
import { defaultLabels, Service, Volume } from "@intentius/chant-lexicon-docker";

export const labels = defaultLabels({
  "com.example.team": "platform",
  "com.example.managed-by": "chant",
});

export const api = new Service({ image: "myapp:1.0" });
export const db = new Service({ image: "postgres:16-alpine" });
// Both api and db inherit the labels above
```
